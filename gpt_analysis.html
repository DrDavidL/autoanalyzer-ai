for col in categorical_columns:
    unique_values = df[col].unique()
    if len(unique_values) == 2:
        # Convert binary categorical columns to numeric
        most_common_value = df[col].mode()[0]
        least_common_value = [val for val in unique_values if val != most_common_value][0]
        df[col] = df[col].map({most_common_value: 0, least_common_value: 1})
